<objective>
Replace adm-zip in-memory ZIP assembly in `routes/download.js` with archiver v7 streaming piped directly to the HTTP response. Add p-limit(8) concurrency pool. Call archive.append() INSIDE the p-limit callback for each clip immediately after ID3 tagging — never accumulate audio buffers in a results array. Embed ID3 tags entirely in-memory (NodeID3.write returns Buffer — no temp files needed). Eliminate temp directory creation. Wire SSE progress events by reading sessionId from the POST body. Wire AbortController to client disconnect so in-flight fetches cancel cleanly.

Purpose: Large playlists currently exhaust Replit VM RAM because adm-zip accumulates the entire ZIP in memory before responding. Archiver streams bytes to the response as each clip is processed. The previous RESEARCH.md draft showed a post-Promise.all append loop — this plan explicitly replaces that with append-inside-callback so buffers are GC'd immediately after archiver writes them to the stream. With p-limit(8) the peak in-flight memory is bounded to ~8 clips at a time (~80-160 MB) regardless of playlist size.

Output: Modified routes/download.js, updated package.json (archiver@^7 + p-limit@^4 added, adm-zip removed).
</objective>

```javascript
router.post('/playlist', async (req, res) => {
  const { playlist, clips, embedImage, sessionId } = req.body;

  if (!playlist || !clips || !Array.isArray(clips) || clips.length === 0) {
    return res.status(400).json({ error: 'Invalid playlist data' });
  }

  req.setTimeout(900000);
  res.setTimeout(900000);

  const zipFilename = filenamify(`${playlist.name}.zip`);

  res.setHeader('Content-Type', 'application/zip');
  res.setHeader('Content-Disposition', `attachment; filename="${zipFilename}"`);

  const archive = archiver('zip', { zlib: { level: 6 } });

  // archive error handler: check headersSent before calling res.status()
  // if headers already flushed, use res.destroy(err) — never res.status(500) after streaming starts
  archive.on('error', (err) => {
    console.error('Archive error:', err);
    if (!res.headersSent) {
      res.status(500).json({ error: 'Archive failed' });
    } else {
      res.destroy(err);
    }
  });

  // AbortController cancels in-flight fetches when client disconnects
  const abortController = new AbortController();

  req.on('close', () => {
    if (!res.writableEnded) {
      console.log('Client disconnected, aborting archive and in-flight fetches');
      archive.abort();
      abortController.abort();
    }
  });

  // pipe BEFORE any append() calls
  archive.pipe(res);

  const limit = pLimit(8);
  let completedCount = 0;

  // CRITICAL: archive.append is called INSIDE each limit() callback immediately after
  // ID3 tagging. There is no results array — each buffer is appended and GC'd as
  // archiver writes it to the response stream.
  await Promise.all(
    clips.map(clip => limit(async () => {
      const fileName = filenamify(`${String(clip.no).padStart(2, '0')} - ${clip.title}.mp3`);

      try {
        const audioRes = await fetch(clip.audio_url, { signal: abortController.signal });
        if (!audioRes.ok) {
          console.error(`Failed to download clip ${clip.id}: ${audioRes.statusText}`);
          completedCount++;
          global.downloadTrackers[sessionId]?.sendProgress({
            progress: Math.round((completedCount / clips.length) * 100),
            completedItem: clip.id,
            error: true
          });
          return;
        }

        let audioBuffer = Buffer.from(await audioRes.arrayBuffer());

        // embed ID3 cover art in-memory — no fs.writeFileSync, no disk I/O
        if (embedImage === 'true') {
          const imgRes = await fetch(clip.image_url, { signal: abortController.signal });
          if (imgRes.ok) {
            const imgBuffer = Buffer.from(await imgRes.arrayBuffer());
            const tags = {
              title: clip.title,
              trackNumber: String(clip.no),
              image: {
                mime: 'image/jpeg',
                type: { id: 3, name: 'front cover' },
                description: 'Cover Art',
                imageBuffer: imgBuffer   // in-memory Buffer, not a file path
              }
            };
            // NodeID3.write returns Buffer | false — if false (malformed audio) keep original
            const tagged = NodeID3.write(tags, audioBuffer);
            if (tagged) audioBuffer = tagged;
          }
        }

        // Append immediately — buffer GC'd after archiver flushes it to the response
        archive.append(audioBuffer, { name: fileName });

        completedCount++;
        global.downloadTrackers[sessionId]?.sendProgress({
          progress: Math.round((completedCount / clips.length) * 100),
          completedItem: clip.id
        });
      } catch (err) {
        if (err.name === 'AbortError') return; // client disconnected — silently exit
        console.error(`Error processing clip ${clip.id}:`, err);
        completedCount++;
        global.downloadTrackers[sessionId]?.sendProgress({
          progress: Math.round((completedCount / clips.length) * 100),
          completedItem: clip.id,
          error: true
        });
      }
    }))
  );

  await archive.finalize();
});
```
