**Change D — Replace the downloadPlaylist() function body** with the batch-aware implementation below. Read the current function body carefully first. Key invariants to enforce:

- setupProgressMonitor is called ONCE before the for loop — if the executor moves it inside the loop, each batch opens a new EventSource and previous cleanup() deletes global.downloadTrackers[sessionId], silently dropping all progress events for batch 2+.
- cleanup() is called in the finally block — NOT inside the loop, NOT inside the catch.
- setIsDownloading(false) and setDownloadLabel('') are BOTH in the finally block — D-04 early return skips code after the loop; without finally these never reset and UI stays locked.
- batchHadError is declared in the outer try scope (before the for loop), reset to false at the top of each loop iteration, set to true inside onProgress when data.error is true, checked after each downloadPlaylistApi call.
- The old bulk-success-flip (`setPlaylistClips(... IPlaylistClipStatus.Success ...)` after downloadPlaylistApi) MUST be removed — per-clip status now comes from SSE events; if left in it overwrites per-clip Error states.
- The old standalone `setIsDownloading(false)` after the try/catch MUST be removed — it is now in finally.

```typescript
const downloadPlaylist = async () => {
  if (!playlistData || !playlistClips) return;

  const selectedClips = playlistClips.filter(c => selectedIds.has(c.id));
  if (selectedClips.length === 0) return;

  // Slice into batches of BATCH_SIZE
  const batches: IPlaylistClip[][] = [];
  for (let i = 0; i < selectedClips.length; i += BATCH_SIZE) {
    batches.push(selectedClips.slice(i, i + BATCH_SIZE));
  }
  const totalBatches = batches.length;

  // D-03: flip ALL selected rows to Processing before batch 1 fires
  setPlaylistClips(prev =>
    prev.map(c =>
      selectedIds.has(c.id) ? { ...c, status: IPlaylistClipStatus.Processing } : c
    )
  );

  checkAndShowDonationModal();
  setDownloadPercentage(0);
  setIsDownloading(true);

  // SSE monitor — set up ONCE for all batches, before the for loop.
  // DO NOT call setupProgressMonitor inside the for loop — each call opens a new
  // EventSource and the previous cleanup() deletes global.downloadTrackers[sessionId],
  // causing all progress events for batch 2+ to be silently dropped.
  let batchHadError = false;
  const cleanup = setupProgressMonitor(sessionId, (data) => {
    if (data.progress) {
      setDownloadPercentage(data.progress);
    }
    if (data.completedItem) {
      // BAT-03: handle error:true payload — failed clips flip to Error, not Success
      const status = data.error
        ? IPlaylistClipStatus.Error
        : IPlaylistClipStatus.Success;
      updateClipStatus(data.completedItem, status);
      scrollToRow(data.completedItem);
      if (data.error) {
        batchHadError = true;
      }
    }
  });

  const settings = {
    embed_images: localStorage.getItem('suno-embed-images') || 'true'
  };

  try {
    for (let batchIdx = 0; batchIdx < totalBatches; batchIdx++) {
      const batchNum = batchIdx + 1;
      batchHadError = false; // reset per batch

      // D-02: show batch label during multi-batch downloads
      if (totalBatches > 1) {
        setDownloadLabel(`Downloading batch ${batchNum} of ${totalBatches}…`);
      }
      setDownloadPercentage(0); // reset progress bar per batch

      const batchClips = batches[batchIdx];

      // Filename: plain for single batch, zero-padded numbered for multi
      const zipName = totalBatches === 1
        ? `${playlistData.name}.zip`
        : `${playlistData.name}-batch-${String(batchNum).padStart(2, '0')}-of-${totalBatches}.zip`;

      try {
        await downloadPlaylistApi(
          playlistData,
          batchClips,
          settings.embed_images === 'true',
          sessionId,
          zipName
        );

        // Check if any clip in this batch had an SSE error event
        if (batchHadError) {
          // D-04: treat batch with per-clip errors as a batch failure
          showError(`Failed to download batch ${batchNum} of ${totalBatches}`);
          const failedIds = new Set(
            batches.slice(batchIdx).flatMap(b => b.map(c => c.id))
          );
          setPlaylistClips(prev =>
            prev.map(c =>
              failedIds.has(c.id) ? { ...c, status: IPlaylistClipStatus.Error } : c
            )
          );
          return;
        }
      } catch (err) {
        // D-04: stop-all on first batch HTTP/network failure
        showError(`Failed to download batch ${batchNum} of ${totalBatches}`);
        const failedIds = new Set(
          batches.slice(batchIdx).flatMap(b => b.map(c => c.id))
        );
        setPlaylistClips(prev =>
          prev.map(c =>
            failedIds.has(c.id) ? { ...c, status: IPlaylistClipStatus.Error } : c
          )
        );
        return;
      }
    }

    showSuccess(
      totalBatches === 1
        ? 'Playlist ZIP file download initiated'
        : `All ${totalBatches} batch ZIPs downloaded`
    );
  } finally {
    // cleanup() closes the EventSource — called once after all batches complete (or on error/return)
    cleanup();
    setIsDownloading(false);
    setDownloadLabel('');
  }
};
```
