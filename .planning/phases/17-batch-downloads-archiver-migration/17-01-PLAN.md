<objective>
Replace adm-zip in-memory ZIP assembly in `routes/download.js` with archiver v7 streaming piped directly to the HTTP response. Add p-limit(8) concurrency pool. Embed ID3 tags entirely in-memory (NodeID3.write returns Buffer — no temp files needed). Eliminate temp directory creation. Wire SSE progress events by reading sessionId from the POST body.

Purpose: Large playlists currently exhaust Replit VM RAM because adm-zip accumulates the entire ZIP in memory before responding. Archiver streams bytes to the response as each clip is processed. With p-limit(8) the peak in-flight memory is bounded to ~8 clips at a time (~80-160 MB) regardless of playlist size.

Output: Modified routes/download.js, updated package.json (archiver@^7 + p-limit@^4 added, adm-zip removed).
</objective>

**New POST /playlist handler structure:**

```javascript
router.post('/playlist', async (req, res) => {
  const { playlist, clips, embedImage, sessionId } = req.body;

  if (!playlist || !clips || !Array.isArray(clips) || clips.length === 0) {
    return res.status(400).json({ error: 'Invalid playlist data' });
  }

  req.setTimeout(900000);
  res.setTimeout(900000);

  const zipFilename = filenamify(`${playlist.name}.zip`);

  res.setHeader('Content-Type', 'application/zip');
  res.setHeader('Content-Disposition', `attachment; filename="${zipFilename}"`);

  const archive = archiver('zip', { zlib: { level: 6 } });

  archive.on('error', (err) => {
    console.error('Archive error:', err);
    if (!res.headersSent) {
      res.status(500).json({ error: 'Archive failed' });
    } else {
      res.destroy(err);
    }
  });

  req.on('close', () => {
    console.log('Client disconnected, aborting archive');
    archive.abort();
    res.destroy();
  });

  archive.pipe(res);

  const limit = pLimit(8);
  let completedCount = 0;

  const results = await Promise.all(
    clips.map(clip => limit(async () => {
      try {
        const fileName = filenamify(`${String(clip.no).padStart(2, '0')} - ${clip.title}.mp3`);

        const audioRes = await fetch(clip.audio_url);
        if (!audioRes.ok) {
          console.error(`Failed to download clip ${clip.id}: ${audioRes.statusText}`);
          completedCount++;
          global.downloadTrackers[sessionId]?.sendProgress({
            progress: Math.round((completedCount / clips.length) * 100),
            completedItem: clip.id,
            error: true
          });
          return null;
        }

        let audioBuffer = Buffer.from(await audioRes.arrayBuffer());

        if (embedImage === 'true') {
          const imgRes = await fetch(clip.image_url);
          if (imgRes.ok) {
            const imgBuffer = Buffer.from(await imgRes.arrayBuffer());
            const tags = {
              title: clip.title,
              trackNumber: String(clip.no),
              image: {
                mime: 'image/jpeg',
                type: { id: 3, name: 'front cover' },
                description: 'Cover Art',
                imageBuffer: imgBuffer
              }
            };
            const tagged = NodeID3.write(tags, audioBuffer);
            if (tagged) audioBuffer = tagged;
          }
        }

        completedCount++;
        global.downloadTrackers[sessionId]?.sendProgress({
          progress: Math.round((completedCount / clips.length) * 100),
          completedItem: clip.id
        });

        return { buffer: audioBuffer, fileName };
      } catch (err) {
        console.error(`Error processing clip ${clip.id}:`, err);
        completedCount++;
        global.downloadTrackers[sessionId]?.sendProgress({
          progress: Math.round((completedCount / clips.length) * 100),
          completedItem: clip.id,
          error: true
        });
        return null;
      }
    }))
  );

  for (const result of results) {
    if (result) {
      archive.append(result.buffer, { name: result.fileName });
    }
  }

  await archive.finalize();
});
```

The try/catch around the outer handler body can be removed since archiver.on('error') handles archive failures and individual clip errors are caught inside the p-limit callbacks. If a top-level error occurs before Promise.all resolves, wrap with a try/catch that checks `res.headersSent`.

Keep the GET /progress/:sessionId route unchanged.
